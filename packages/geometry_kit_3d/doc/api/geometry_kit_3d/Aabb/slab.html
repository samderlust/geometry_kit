Ray-vs-box slab test. Returns the entry and exit distances, or null when the ray misses or the box is entirely behind the ray.

Implementation

static (double, double)? slab(Ray3 ray, Point3 min, Point3 max) {
  var t0 = double.negativeInfinity, t1 = double.infinity;
  for (var k = 0; k < 3; k++) {
    final o = ray.origin[k], d = ray.direction[k];
    if (d.abs() < 1e-15) {
      if (o < min[k] || o > max[k]) return null;
      continue;
    }
    var a = (min[k] - o) / d, b = (max[k] - o) / d;
    if (a > b) (a, b) = (b, a);
    if (a > t0) t0 = a;
    if (b < t1) t1 = b;
    if (t0 > t1) return null;
  }
  if (t1 < 0) return null;
  return (t0, t1);
}
